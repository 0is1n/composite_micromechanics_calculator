""" Composite Micromechanics Calculator Two modes: "lamina" - stress and strain in xy and 12 systems "laminate" - lamina matrices, ABD and its inverse, mid-plane strain + curvature, stress/strain at a point at distance z from NA Units: moduli in GPa, stress in MPa, strain in microstrain, lengths in mm, force flow N/mm, moment flow N*mm/mm, curvature 1/mm Conventions: - lamina are counted from the top down - z measured from NA, downward +ve - theta +ve from x-axis to 1-axis """ import numpy as np # ------------------------- INPUTS ------------------------- E1, E2, mu12, G12 = 19.755, 1.9755, 0.35, 0.7 # GPa mode = "laminate" # "lamina" or "laminate" # ---- lamina ---- theta_deg = 60 # degrees known = "stress" # "stress" or "strain" system = "xy" # coordinate system: "xy" or "12" load = [-3.5, 7, -1.4] # stress[MPa] or strain[microstrain] # ---- laminate ---- layup = [45, 0] # lamina orientations symmetric = False # true of false t = [3, 5] # lamina thickness (one value or list) lam_known = "load" # "load" (q, m) or "strain" (eps0, kappa) N = [0, 0, 0] # [qx, qy, qxy] force flow [N/mm] M = [10, 0, 0] # [mx, my, mxy] moment flow [N*mm/mm] eps0 = [0, 0, 0] # [ex, ey, gxy] mid-plane strain [microstrain]) kappa = [0, 0, 0] # [kx, ky, kxy] curvature [1/mm] z_point = -2.5 # distance from mid-plane [mm] # ----------------------------------------------------------------------- def transforms(theta_deg): """Transformation matrix T, xy to 12""" t = np.radians(theta_deg) m, n = np.cos(t), np.sin(t) Ts = np.array([[m * m, n * n, 2 * m * n], # for stress T [n * n, m * m, -2 * m * n], [-m * n, m * n, m * m - n * n]]) Te = np.array([[m * m, n * n, m * n], #for strain T^T^-1 [n * n, m * m, -m * n], [-2 * m * n, 2 * m * n, m * m - n * n]]) return Ts, Te def material_SQ(E1, E2, mu12, G12): """Compliance S and stiffness Q matricies in 12 system""" mu21 = mu12 * E2 / E1 d = 1 - mu12 * mu21 S = np.array([[1 / E1, -mu12 / E1, 0], [-mu12 / E1, 1 / E2, 0], [0, 0, 1 / G12]]) Q = np.array([[E1 / d, mu12 * E2 / d, 0], [mu12 * E2 / d, E2 / d, 0], [0, 0, G12]]) return S, Q def global_SQ(S, Q, theta_deg): """Qbar and Sbar in xy system""" Ts, Te = transforms(theta_deg) Qb = np.linalg.inv(Ts) @ Q @ Te Sb = np.linalg.inv(Qb) return Qb, Sb def lamina_solver(known, system, load, S, Q, theta_deg): """Lamina: return stress [MPa] and strain in both systems""" Ts, Te = transforms(theta_deg) Qb, Sb = global_SQ(S, Q, theta_deg) Qb, Sb = Qb * 1e3, Sb / 1e3 # GPa to MPa v = np.array(load, dtype=float) if known == "strain": v = v * 1e-6 # microstrain to strain exy = v if system == "xy" else np.linalg.inv(Te) @ v sxy = Qb @ exy elif known == "stress": sxy = v if system == "xy" else np.linalg.inv(Ts) @ v exy = Sb @ sxy else: raise ValueError("known must be either 'stress' or 'strain'") return sxy, Ts @ sxy, exy, Te @ exy # -------------------- Laminate Setup ---------------------- def build_layup(layup, symmetric, t): """Turn lamina list into geometry (angles, thicknesses, h)""" angles = [float(a) for a in layup] # layup angles list th = [float(x) for x in t] if np.ndim(t) else [float(t)] * len(angles) # layup thickness list if len(th) != len(angles): raise ValueError("t must be one number or one t per lamina") if symmetric: angles = angles + angles[::-1] th = th + th[::-1] H = sum(th) # laminate thickness h = -H / 2 + np.concatenate([[0], np.cumsum(th)]) # lamina boundaries return angles, th, h, H def laminate_ABD(Q, angles, h): """Matricies: A [N/mm], B [N], D [N mm]""" A, B, D = np.zeros((3, 3)), np.zeros((3, 3)), np.zeros((3, 3)) # initialise matricies for ang, h0, h1 in zip(angles, h[:-1], h[1:]): Qb = global_SQ(None, Q, ang)[0] * 1e3 # GPa to MPa A += Qb * (h1 - h0) B += Qb * (h1 ** 2 - h0 ** 2) / 2 D += Qb * (h1 ** 3 - h0 ** 3) / 3 ABD = np.block([[A, B], [B, D]]) return A, B, D, ABD def engineering_constants(a, d, H): """ These are equivalent Young's Moduli of the laminate when B = 0. a, d = compliance matricies parts, H = laminate thickness ten = tensile, ben = bending (Ex, Ey, Gxy, muxy, muyx, mx, my (transverse coefficients)) """ ten = dict(Ex=1 / (H * a[0, 0]), Ey=1 / (H * a[1, 1]), Gxy=1 / (H * a[2, 2]), mu_xy=-a[0, 1] / a[0, 0], mu_yx=-a[0, 1] / a[1, 1]) ben = dict(Ex=12 / (H ** 3 * d[0, 0]), Ey=12 / (H ** 3 * d[1, 1]), Gxy=12 / (H ** 3 * d[2, 2]), mu_xy=-d[0, 1] / d[0, 0], mu_yx=-d[0, 1] / d[1, 1]) return ten, ben # moduli in MPa def laminae_at(z, zb): """Index of the lamina at height z""" return [int(np.clip(np.searchsorted(zb, z) - 1, 0, len(zb) - 2))] # clip so if z=-H/2 you dont get index -1 def lamina_state(Q, ang, eps0, kappa, z): """Strain and stress in a lamina at height z returned as a dictionary (strain, stress in MPa)""" Ts, Te = transforms(ang) Qb = global_SQ(None, Q, ang)[0] * 1e3 exy = eps0 + z * kappa sxy = Qb @ exy return dict(exy=exy, e12=Te @ exy, sxy=sxy, s12=Ts @ sxy) # ------------------------------------------ def laminate_solver(lam_known, N, M, eps0, kappa, ABD): """Return mid-plane strain and curvature [-] and [1/mm]""" if lam_known == "load": sol = np.linalg.solve(ABD, np.array(list(N) + list(M), dtype=float)) return sol[:3], sol[3:] if lam_known == "strain": return np.array(eps0, dtype=float) * 1e-6, np.array(kappa, dtype=float) def show(name, M, sci=False): M = np.array(M, dtype=float) if sci: M = np.where(np.abs(M) < 1e-12 * np.abs(M).max(), 0, M) # drop round-off noise fmt = {"float_kind": lambda x: f"{x: .3e}"} if sci else None print(f"\n{name}\n{np.array2string(M, precision=4, suppress_small=True, formatter=fmt)}") S, Q = material_SQ(E1, E2, mu12, G12) if mode == "lamina": Qb, Sb = global_SQ(S, Q, theta_deg) show("Q [GPa]", Q) show("S [1/GPa]", S) show("Ts (stress xy to 12)", transforms(theta_deg)[0]) show("Te (strain xy to 12)", transforms(theta_deg)[1]) show("Qbar [GPa]", Qb) show("Sbar [1/GPa]", Sb) sxy, s12, exy, e12 = lamina_solver(known, system, load, S, Q, theta_deg) show("stress_xy [MPa]", sxy) show("stress_12 [MPa]", s12) show("strain_xy [microstrain]", exy * 1e6) show("strain_12 [microstrain]", e12 * 1e6) elif mode == "laminate": angles, th, hb, H = build_layup(layup, symmetric, t) print(f"Layup: {angles}") print(f"Lamina thickness [mm]: {th} total H = {H:g} mm") print(f"Lamina boundaries h [mm] from NA: {np.round(hb, 6).tolist()}") show("Q [GPa] (same for every lamina)", Q) show("S [1/GPa] (same for every lamina)", S) for i, ang in enumerate(angles, 1): Ts, Te = transforms(ang) Qb, Sb = global_SQ(S, Q, ang) print(f"\n================ lamina {i}: theta = {ang:g} deg, z = {hb[i-1]:g} to {hb[i]:g} mm ================") show("Ts (stress xy to 12)", Ts) show("Te (strain xy to 12)", Te) show("Qbar [GPa]", Qb) show("Sbar [1/GPa]", Sb) A, B, D, ABD = laminate_ABD(Q, angles, hb) print("\n================ LAMINATE ================") show("A [N/mm]", A) show("B [N]", B) show("D [N mm]", D) show("ABD [[A,B],[B,D]]", ABD) abd = np.linalg.inv(ABD) show("abd = inv(ABD) (units: 1/(N/mm), 1/N, 1/(N mm))", abd, sci=True) ten, ben = engineering_constants(abd[:3, :3], abd[3:, 3:], H) print("\nEquivalent constants (only for symmetric laminates)") for label, c in (("tensile", ten), ("bending", ben)): print(f" {label}: Ex = {c['Ex'] * 1e-3:.4f} GPa, Ey = {c['Ey'] * 1e-3:.4f} GPa, Gxy = {c['Gxy'] * 1e-3:.4f} GPa, " f"mu_xy = {c['mu_xy']:.4f}, mu_yx = {c['mu_yx']:.4f}") e0, kap = laminate_solver(lam_known, N, M, eps0, kappa, ABD) if lam_known == "strain": NM = ABD @ np.concatenate([e0, kap]) show("N [N/mm]", NM[:3]) show("M [N mm/mm]", NM[3:]) show("mid-plane strain eps0 [microstrain]", e0 * 1e6) show("curvature kappa [micro 1/mm]", kap * 1e6) print(f"\n================ POINT AT z = {z_point:g} mm ================") for k in laminae_at(z_point, hb): r = lamina_state(Q, angles[k], e0, kap, z_point) print(f"\n--- lamina {k + 1} (theta = {angles[k]:g} deg) ---") show("strain_xy [microstrain]", r["exy"] * 1e6) show("strain_12 [microstrain]", r["e12"] * 1e6) show("stress_xy [MPa]", r["sxy"]) show("stress_12 [MPa]", r["s12"]) print("\n================ STRESS 12 AT TOP / BOTTOM OF EACH lamina [MPa] ================") print(f"{'lamina':>4} {'theta':>7} {'h':>9} {'sigma1':>11} {'sigma2':>11} {'tau12':>11}") for i, ang in enumerate(angles): for zz in (hb[i], hb[i + 1]): s = lamina_state(Q, ang, e0, kap, zz)["s12"] print(f"{i + 1:>4} {ang:>7g} {zz:>9.4f} {s[0]:>11.4f} {s[1]:>11.4f} {s[2]:>11.4f}")
